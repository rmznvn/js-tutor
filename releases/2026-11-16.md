// Интервалы заданы минутами от начала дня: 600 = 10:00.
const slots = [
  { id: 1, start: 600, end: 630 },
  { id: 2, start: 630, end: 660 },
  { id: 3, start: 660, end: 690 },
  { id: 4, start: 690, end: 720 },
];
const bookings = [{ start: 630, end: 660 }];

function getAvailableSlots(items, busy, minDuration) {
  // filter создаёт новый массив; исходные интервалы остаются неизменными.
  return items.filter((slot) => {
    // Ровно нужная длительность подходит: используем >=, а не >.
    const longEnough = slot.end - slot.start >= minDuration;
    // some отвечает на вопрос «есть хотя бы одно пересечение?».
    // Строгие сравнения разрешают соседние интервалы: [600,630) и [630,660).
    const overlaps = busy.some((booking) =>
      slot.start < booking.end && booking.start < slot.end
    );
    // Оставляем только достаточно длинные слоты без пересечений.
    return longEnough && !overlaps;
  }).sort((a, b) => a.start - b.start); // Сортируем результат filter, не items.
}

function formatTime(minutes) {
  // Целая часть — часы, остаток — минуты; padStart добавляет ведущий ноль.
  return String(Math.floor(minutes / 60)).padStart(2, "0") + ":" +
    String(minutes % 60).padStart(2, "0");
}

const input = document.getElementById("duration");
const list = document.getElementById("slots");
const message = document.getElementById("message");
function render() {
  // value у select — строка. Преобразуем её перед числовым сравнением.
  const visible = getAvailableSlots(slots, bookings, Number(input.value));
  list.replaceChildren();
  message.textContent = visible.length ? "Свободных слотов: " + visible.length : "Нет подходящих слотов";
  visible.forEach((slot) => {
    const li = document.createElement("li");
    // Данные выводим текстом: HTML из названий сюда не подставляется.
    li.textContent = formatTime(slot.start) + "–" + formatTime(slot.end);
    list.append(li);
  });
}
input.addEventListener("change", render);
render();

```jsx
import { useState } from "react";

const initialProfile = { name: "Наталья", role: "Webflow developer" };

export default function ProfileEditor() {
  // Сохранённые данные и редактируемый черновик имеют разный смысл.
  const [saved, setSaved] = useState(initialProfile);
  const [draft, setDraft] = useState({ ...initialProfile });
  const [message, setMessage] = useState("");
  // Сравниваем содержимое полей, а не ссылки объектов: это производное значение.
  const dirty = draft.name !== saved.name || draft.role !== saved.role;
  // trim нужен для проверки имени из одних пробелов.
  const error = draft.name.trim() ? "" : "Введите имя";

  function update(field, value) {
    // Копия сохраняет другое поле. Функциональное обновление берёт актуальный draft.
    setDraft((current) => ({ ...current, [field]: value }));
    setMessage("");
  }
  function save(event) {
    event.preventDefault(); // Обрабатываем submit без перезагрузки страницы.
    if (error || !dirty) return; // Повторяем проверку внутри обработчика.
    const next = { name: draft.name.trim(), role: draft.role.trim() };
    setSaved(next); // Фиксируем подтверждённые значения.
    setDraft({ ...next }); // Убираем отличия, вызванные нормализацией пробелов.
    setMessage("Сохранено в памяти страницы"); // Здесь нет запроса к серверу.
  }
  function cancel() {
    setDraft({ ...saved }); // Возвращаем именно последнее сохранение, не initialProfile.
    setMessage("Изменения отменены");
  }
  return (
    <main>
      <h1>Редактор профиля</h1>
      <p>Сохранено: {saved.name} — {saved.role || "Роль не указана"}</p>
      <form onSubmit={save} noValidate>
        <label htmlFor="name">Имя</label>
        {/* Управляемое поле отображает draft; saved меняется только по submit. */}
        <input id="name" value={draft.name} onChange={(e) => update("name", e.target.value)} aria-invalid={Boolean(error)} aria-describedby="name-error" />
        <p id="name-error">{error}</p>
        <label htmlFor="role">Роль (необязательно)</label>
        <input id="role" value={draft.role} onChange={(e) => update("role", e.target.value)} />
        <p>{dirty ? "Есть несохранённые изменения" : "Все изменения сохранены"}</p>
        {/* disabled — удобство интерфейса; проверка в save остаётся обязательной. */}
        <button type="submit" disabled={!dirty || Boolean(error)}>Сохранить</button>
        <button type="button" disabled={!dirty} onClick={cancel}>Отменить</button>
      </form>
      <p role="status">{message}</p>
    </main>
  );
}
```

      <label><input id="free" type="checkbox" checked={filters.onlyFree} onChange={(e) => change("onlyFree", e.target.checked)} />Только бесплатные</label>
      <div aria-label="Активные фильтры">
        {/* После удаления чипа управляемое поле само отображает сброшенное значение. */}
        {filters.category && <button type="button" onClick={() => remove("category")}>Убрать фильтр: {filters.category === "design" ? "Дизайн" : "Код"}</button>}
        {filters.onlyFree && <button type="button" onClick={() => remove("onlyFree")}>Убрать фильтр: Бесплатные</button>}
      </div>
      <p role="status">{visible.length ? "Найдено: " + visible.length : "Ничего не найдено"}</p>
      <ul>{visible.map((item) => <li key={item.id}>{item.title}</li>)}</ul>
    </main>
  );
}
